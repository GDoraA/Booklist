/**
 * A Líra a saját alkalmazásazonosító User-Agentet Cloudflare-oldallal
 * blokkolja, ezért a keresési és termékoldalakat valós böngészőhöz
 * hasonló fejlécekkel kérjük le.
 */
function getLiraFetchOptions_() {
  return {
    muteHttpExceptions: true,
    followRedirects: true,
    headers: {
      "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) " +
        "AppleWebKit/537.36 (KHTML, like Gecko) Chrome/140.0.0.0 Safari/537.36",
      "Accept": "text/html,application/xhtml+xml,application/xml;q=0.9,image/avif,image/webp,*/*;q=0.8",
      "Accept-Language": "hu-HU,hu;q=0.9,en;q=0.7",
      "Referer": "https://www.lira.hu/hu/cimlap"
    }
  };
}

function fetchLiraHtml_(url) {
  const response = UrlFetchApp.fetch(url, getLiraFetchOptions_());

  const code = response.getResponseCode();
  const html = response.getContentText("UTF-8");
  if (code < 200 || code >= 300) {
    throw new Error("Líra oldal letöltési hiba: " + code + " - " + url);
  }
  if (/Sorry, you have been blocked|Attention Required!\s*\|\s*Cloudflare/i.test(html)) {
    throw new Error("A Líra Cloudflare-védelme blokkolta a lekérést.");
  }
  return html;
}

/**
 * A hirdetési és követési paraméterek (gclid, gbraid, kampany stb.)
 * gyakran Cloudflare 403-at okoznak szerveroldali lekérésnél. A Líra
 * termékazonosítója az útvonalban van, ezért a query és fragment elhagyható.
 */
function canonicalizeLiraProductUrl_(url) {
  const cleanUrl = decodeHtmlEntities_(String(url || "").trim())
    .split("#")[0]
    .split("?")[0]
    .replace(/^http:\/\/(?:www\.)?lira\.hu/i, "https://www.lira.hu")
    .replace(/^https:\/\/lira\.hu/i, "https://www.lira.hu");

  return cleanUrl;
}

/** A `Cím:`, `Szerző:` stb. felirat melletti span.data értékét olvassa ki. */
function extractLiraDataField_(html, fieldName) {
  const escapedFieldName = String(fieldName || "")
    .replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

  if (!escapedFieldName) return "";

  return extractFirstMatch_(
    html,
    new RegExp(
      "<span[^>]*class=[\\\"'][^\\\"']*\\btxt\\b[^\\\"']*[\\\"'][^>]*>" +
      "\\s*" + escapedFieldName + "\\s*:?\\s*</span>\\s*" +
      "<span[^>]*class=[\\\"'][^\\\"']*\\bdata\\b[^\\\"']*[\\\"'][^>]*>" +
      "([\\s\\S]*?)</span>",
      "i"
    )
  );
}

/** Csak a kampányjellegű szögletes zárójeles utótagokat távolítja el. */
function cleanLiraTitle_(title) {
  return String(title || "")
    .replace(/\s*\[[^\]]*akci[oó][^\]]*\]\s*$/i, "")
    .trim();
}

function extractLiraProductCode_(html) {
  return extractFirstMatch_(
    html,
    /<input[^>]*name=["']product-akod["'][^>]*value=["']([^"']+)["'][^>]*>/i
  ) || extractFirstMatch_(
    html,
    /<input[^>]*value=["']([^"']+)["'][^>]*name=["']product-akod["'][^>]*>/i
  );
}

function extractLiraDebugContext_(html, pattern) {
  const source = String(html || "");
  const match = source.match(pattern);
  if (!match) return "";

  const index = typeof match.index === "number" ? match.index : 0;
  return source
    .substring(Math.max(0, index - 180), Math.min(source.length, index + 520))
    .replace(/\s+/g, " ")
    .trim();
}

/**
 * Részletes, nem destruktív diagnosztika egy közvetlen Líra-termékoldalhoz.
 * Megmutatja a hálózati választ, a blokkolási jeleket, a releváns HTML-
 * környezetet, a kinyert mezőket és azt is, elfogadná-e az árfigyelés.
 */
function debugLiraProductPage_(productUrl) {
  const requestedUrl = String(productUrl || "").trim();
  const canonicalUrl = canonicalizeLiraProductUrl_(requestedUrl);
  const debug = {
    requestedUrl: requestedUrl,
    canonicalUrl: canonicalUrl,
    fetch: {},
    signals: {},
    fields: {},
    prices: {},
    snippets: {},
    wishlistAcceptance: {}
  };

  if (!canonicalUrl) {
    debug.fetch.error = "Nincs megadva Líra-termékoldal URL.";
    return debug;
  }

  let response;
  let html = "";
  try {
    response = UrlFetchApp.fetch(canonicalUrl, getLiraFetchOptions_());
    debug.fetch.responseCode = response.getResponseCode();
    debug.fetch.headers = response.getAllHeaders ? response.getAllHeaders() : response.getHeaders();
    html = response.getContentText("UTF-8");
    debug.fetch.htmlLength = html.length;
  } catch (err) {
    debug.fetch.error = err && err.message ? err.message : String(err);
    return debug;
  }

  debug.signals.cloudflareBlocked = /Sorry, you have been blocked|Attention Required!\s*\|\s*Cloudflare/i.test(html);
  debug.signals.hasOnlinePriceClass = /class=["'][^"']*\bonline-price\b/i.test(html);
  debug.signals.hasSalePriceClass = /class=["'][^"']*\bsale-price\b(?!-)/i.test(html);
  debug.signals.hasBookPriceActionClass = /class=["'][^"']*\bbook-price-action\b/i.test(html);
  debug.signals.hasPriceLabel = /online\s*ár\s*:/i.test(decodeHtmlEntities_(stripHtml_(html)));
  debug.signals.hasFullPriceLabel = /ár\s+a\s+könyvön\s*:/i.test(decodeHtmlEntities_(stripHtml_(html)));

  const rawTitle = extractLiraDataField_(html, "Cím");
  const rawPublicationDate = extractLiraDataField_(html, "Megjelenés");
  debug.fields = {
    titleRaw: rawTitle,
    title: cleanLiraTitle_(rawTitle),
    productCode: extractLiraProductCode_(html),
    authorRaw: extractLiraDataField_(html, "Szerző"),
    publisher: extractLiraDataField_(html, "Kiadó"),
    series: extractLiraDataField_(html, "Sorozat"),
    publicationDateRaw: rawPublicationDate,
    publicationDate: normalizePublicationDate_(rawPublicationDate),
    isbn: String(extractLiraDataField_(html, "ISBN") || "").replace(/[^0-9Xx]/g, "")
  };

  const commerceOffer = extractCommerceOffer_(html);
  const plainText = decodeHtmlEntities_(stripHtml_(html))
    .replace(/\u00a0/g, " ")
    .replace(/\s+/g, " ");
  const debugOnlinePrice = extractLiraOnlinePrice_(html);
  const debugFullPrice = extractLiraFullPrice_(html);
  const debugEffectivePrice = debugOnlinePrice || commerceOffer.price || debugFullPrice;
  debug.prices = {
    onlinePrice: debugOnlinePrice,
    fullPrice: debugFullPrice,
    schemaOrLabeledPrice: commerceOffer.price,
    schemaAvailability: commerceOffer.availability,
    shipping: (plainText.match(/Szállítás:\s*([0-9]+\s*-\s*[0-9]+\s*munkanap)/i) || [])[1] || "",
    effectivePrice: debugEffectivePrice,
    effectivePriceSource: debugOnlinePrice
      ? "online-price vagy sale-price"
      : commerceOffer.price
        ? "schema.org / címkézett ajánlati ár"
        : debugFullPrice
          ? "teljes ár tartalék"
          : "nincs"
  };
  debug.snippets = {
    onlinePrice: extractLiraDebugContext_(html, /class=["'][^"']*\bonline-price\b/i),
    salePrice: extractLiraDebugContext_(html, /class=["'][^"']*\bsale-price\b(?!-)/i),
    fullPrice: extractLiraDebugContext_(html, /ár\s+a\s+könyvön\s*:/i),
    productData: extractLiraDebugContext_(html, />\s*Cím\s*:/i)
  };

  debug.wishlistAcceptance.accepted = Boolean(debug.prices.effectivePrice);
  debug.wishlistAcceptance.reason = debug.wishlistAcceptance.accepted
    ? "Az effektív ár kitöltheti a candidate.price mezőt: " + debug.prices.effectivePriceSource + "."
    : "Sem online, strukturált ajánlati, sem teljes ár nem volt kinyerhető.";

  return debug;
}

function extractLiraOnlinePrice_(html) {
  const priceElement = extractFirstMatch_(
    html,
    /<div[^>]*class=["'](?:[^"']*\s)?online-price(?:\s[^"']*)?["'][^>]*>([\s\S]*?)<\/div>/i
  );
  if (priceElement) {
    const elementMatch = String(priceElement).match(
      /([0-9]{1,3}(?:[ .][0-9]{3})+|[0-9]{3,})/
    );
    if (elementMatch) return elementMatch[1].replace(/[ .]/g, "");
  }

  // Ha nincs normál online ár, az időszakos akciós ár a sale-price
  // elemben jelenik meg. Ez szándékosan csak másodlagos forrás.
  const salePriceElement = extractFirstMatch_(
    html,
    /<div[^>]*class=["'][^"']*\bsale-price\b(?!-)[^"']*["'][^>]*>([\s\S]*?)<\/div>/i
  );
  if (salePriceElement) {
    const salePriceMatch = String(salePriceElement).match(
      /([0-9]{1,3}(?:[ .][0-9]{3})+|[0-9]{1,})\s*(?:Ft|HUF)?/i
    );
    if (salePriceMatch) return salePriceMatch[1].replace(/[ .]/g, "");
  }

  const text = decodeHtmlEntities_(stripHtml_(html))
    .replace(/\u00a0/g, " ")
    .replace(/\s+/g, " ");
  const match = text.match(
    /online\s*ár\s*:[\s\S]{0,900}?([0-9]{1,3}(?:[ .][0-9]{3})+|[0-9]{3,})\s*Ft/i
  );
  return match ? match[1].replace(/[ .]/g, "") : "";
}

function extractLiraFullPrice_(html) {
  const structuredPrice = extractFirstMatch_(
    html,
    /<div[^>]*class=["'][^"']*\bbook-price-action\b[^"']*["'][^>]*>[\s\S]*?ár\s+a\s+könyvön\s*:[\s\S]*?<\/div>\s*<div[^>]*class=["'][^"']*\bbook-price-action\b[^"']*["'][^>]*>\s*([0-9]{1,3}(?:[ .][0-9]{3})*|[0-9]+)\s*Ft/i
  );
  if (structuredPrice) return structuredPrice.replace(/[ .]/g, "");

  const text = decodeHtmlEntities_(stripHtml_(html))
    .replace(/\u00a0/g, " ")
    .replace(/\s+/g, " ");
  const match = text.match(
    /ár\s*a\s*könyvön\s*:[\s\S]{0,500}?([0-9]{1,3}(?:[ .][0-9]{3})+|[0-9]{3,})\s*Ft/i
  );
  return match ? match[1].replace(/[ .]/g, "") : extractFullCommercePrice_(html);
}

function parseLiraProductPage_(productUrl) {
  const canonicalProductUrl = canonicalizeLiraProductUrl_(productUrl);
  const productHtml = fetchLiraHtml_(canonicalProductUrl);
  const plainText = decodeHtmlEntities_(stripHtml_(productHtml))
    .replace(/\u00a0/g, " ")
    .replace(/\s+/g, " ");
  const title = cleanLiraTitle_(
    extractLiraDataField_(productHtml, "Cím") ||
    extractFirstMatch_(productHtml, /<h1[^>]*>([\s\S]*?)<\/h1>/i) ||
    extractFirstMatch_(productHtml, /<meta property=["']og:title["'] content=["']([^"']+)/i)
  );
  const author = normalizeAuthorForSearchOrder_(
    extractLiraDataField_(productHtml, "Szerző") ||
    extractFirstMatch_(productHtml, /<h2[^>]*>[\s\S]*?<a[^>]*>([\s\S]*?)<\/a>/i) ||
    extractFirstMatch_(productHtml, /Szerző:\s*[\s\S]{0,120}?<a[^>]*>([\s\S]*?)<\/a>/i)
  );
  const publisher =
    extractLiraDataField_(productHtml, "Kiadó") ||
    extractFirstMatch_(productHtml, /Kiadó:\s*[\s\S]{0,120}?<a[^>]*>([\s\S]*?)<\/a>/i);
  const series = extractLiraDataField_(productHtml, "Sorozat");
  const publicationDateRaw = extractLiraDataField_(productHtml, "Megjelenés");
  const publicationDate = normalizePublicationDate_(publicationDateRaw);
  const year = publicationDate
    ? publicationDate.substring(0, 4)
    : (String(publicationDateRaw || "").match(/\b(1[0-9]{3}|20[0-9]{2}|2100)\b/) || [])[1] || "";
  const foundIsbn = String(
    extractLiraDataField_(productHtml, "ISBN") ||
    (plainText.match(/ISBN:\s*([0-9Xx-]{10,20})/i) || [])[1] || ""
  ).replace(/[^0-9Xx]/g, "");
  const productCode = extractLiraProductCode_(productHtml);
  const commerceOffer = extractCommerceOffer_(productHtml);
  const onlinePrice = extractLiraOnlinePrice_(productHtml);
  const fullPrice = extractLiraFullPrice_(productHtml);
  const price = onlinePrice || commerceOffer.price || fullPrice;
  const shipping = (plainText.match(/Szállítás:\s*([0-9]+\s*-\s*[0-9]+\s*munkanap)/i) || [])[1] || "";
  const imageUrl = extractFirstMatch_(productHtml, /<meta property=["']og:image["'] content=["']([^"']+)/i);

  if (!title) return null;
  return {
    source: "lira", sourceId: canonicalProductUrl, productCode: productCode,
    title: title, authors: author, originalTitle: "", previousTitle: "",
    series: series, number: "", year: year, publicationDate: publicationDate,
    location: "", shelf: "", pageCount: "", isbn: foundIsbn,
    publisher: publisher, translator: "", genre: "",
    coverUrl: imageUrl, language: "hu",
    price: price, onlinePrice: onlinePrice || commerceOffer.price || "", fullPrice: fullPrice,
    availability: commerceOffer.availability || (price ? "Rendelhető" : "Nincs adat"),
    shipping: shipping, productUrl: canonicalProductUrl
  };
}

/**
 * Líra keresési találati oldalból könyv-termékoldal URL-eket gyűjt.
 */
function extractLiraProductUrls_(html) {
  const matches = String(html || "").match(
    /(?:https:\/\/(?:www\.)?lira\.hu)?\/hu\/konyv\/[^"' <#?]+/ig
  ) || [];
  const seen = {};

  return matches.map(function(url) {
    const decodedUrl = decodeHtmlEntities_(url);
    return decodedUrl.indexOf("/hu/konyv/") === 0
      ? "https://www.lira.hu" + decodedUrl
      : decodedUrl;
  }).filter(function(url) {
    if (seen[url]) return false;
    seen[url] = true;
    return true;
  });
}

function buildLiraSearchUrl_(query) {
  return "https://www.lira.hu/hu/reszletes_kereso" +
    "?listorder=release_date&listdirection=desc" +
    "&listpagenumber=50&listcurrentpage=0&quick_search_txt=" +
    encodeURIComponent(String(query || "").trim()) +
    "&quick_search_category_code=";
}

/** A Líra gyorskeresőhöz rövid, címközpontú keresőkifejezést készít. */
function buildLiraTitleSearchQuery_(title) {
  return String(title || "")
    .trim()
    .replace(/^(?:a|az|egy)\s+/i, "")
    .replace(/\s+/g, " ")
    .toLowerCase();
}

/** A Líra gyorskereső teljes folyamatának kézzel futtatható diagnosztikája. */
function debugLiraSearch_(query, title, author) {
  const cleanQuery = String(query || "").trim();
  const result = {
    query: cleanQuery,
    searchUrl: buildLiraSearchUrl_(cleanQuery),
    htmlLength: 0,
    blocked: false,
    productUrls: [],
    parsedItems: [],
    errors: []
  };

  if (!cleanQuery) {
    result.errors.push("Üres keresőkifejezés.");
    return result;
  }

  let searchHtml = "";
  try {
    searchHtml = fetchLiraHtml_(result.searchUrl);
    result.htmlLength = searchHtml.length;
    result.blocked = /Sorry, you have been blocked|Attention Required!\s*\|\s*Cloudflare/i.test(searchHtml);
    result.productUrls = extractLiraProductUrls_(searchHtml);
  } catch (err) {
    result.errors.push("Keresőoldal: " + (err && err.message ? err.message : String(err)));
    return result;
  }

  result.productUrls.slice(0, 8).forEach(function(productUrl) {
    try {
      const item = parseLiraProductPage_(productUrl);
      result.parsedItems.push({
        productUrl: productUrl,
        title: item && item.title || "",
        authors: item && item.authors || "",
        isbn: item && item.isbn || "",
        price: item && item.price || "",
        onlinePrice: item && item.onlinePrice || "",
        fullPrice: item && item.fullPrice || "",
        relevant: Boolean(item && isRelevantTitleAuthorMatch_(item, title, author))
      });
    } catch (err) {
      result.errors.push(productUrl + ": " + (err && err.message ? err.message : String(err)));
    }
  });

  return result;
}

function slugifyLiraTitle_(title) {
  return String(title || "")
    .trim()
    .toLowerCase()
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "")
    .replace(/-+/g, "-");
}

/**
 * Ha a Líra keresőoldalát a Cloudflare blokkolja, néhány gyakori
 * könyvkategória kanonikus címútvonalát próbáljuk meg. Az újabb Líra
 * termékeknél a `-1` végződés a leggyakoribb, ezért az kerül előre.
 */
function searchLiraByDirectCandidates_(title, author, trace) {
  const slug = slugifyLiraTitle_(title);
  if (!slug) return [];

  const categoryPaths = [
    "/hu/konyv/szepirodalom/",
    "/hu/konyv/szepirodalom/sci-fi-fantasy-krimi/",
    "/hu/konyv/szepirodalom/felnottirodalom/regenyek/",
    "/hu/konyv/ismeretterjeszto-1/",
    "/hu/konyv/tovabbi-konyveink/"
  ];
  const suffixes = ["-1", ""];

  for (let pathIndex = 0; pathIndex < categoryPaths.length; pathIndex++) {
    for (let suffixIndex = 0; suffixIndex < suffixes.length; suffixIndex++) {
      const url = "https://www.lira.hu" + categoryPaths[pathIndex] + slug + suffixes[suffixIndex];
      try {
        const item = parseLiraProductPage_(url);
        if (trace) trace.directCandidateAttempts.push({
          url: url,
          parsed: Boolean(item),
          relevant: Boolean(item && isRelevantTitleAuthorMatch_(item, title, author)),
          price: item && item.price || ""
        });
        if (item && item.price && isRelevantTitleAuthorMatch_(item, title, author)) {
          return [item];
        }
      } catch (err) {
        if (trace) trace.directCandidateAttempts.push({
          url: url,
          parsed: false,
          error: err && err.message ? err.message : String(err)
        });
        // 404, 403 vagy nem megfelelő útvonal esetén próbáljuk a következőt.
      }
    }
  }
  return [];
}

function searchLira_(query, title, author, trace) {
  if (!String(query || "").trim()) return [];

  if (trace) {
    trace.query = String(query || "").trim();
    trace.searchUrl = buildLiraSearchUrl_(query);
    trace.searchHtmlLength = 0;
    trace.productUrls = [];
    trace.productErrors = [];
    trace.directCandidateAttempts = [];
  }

  let searchHtml = "";
  try {
    searchHtml = fetchLiraHtml_(buildLiraSearchUrl_(query));
    if (trace) trace.searchHtmlLength = searchHtml.length;
  } catch (err) {
    if (trace) trace.searchError = err && err.message ? err.message : String(err);
    if (title) {
      if (trace) trace.usedDirectCandidateFallback = true;
      return searchLiraByDirectCandidates_(title, author, trace);
    }
    return [];
  }
  const productUrls = extractLiraProductUrls_(searchHtml);
  if (trace) trace.productUrls = productUrls.slice(0, 20);
  const results = [];

  productUrls.slice(0, 8).forEach(function(productUrl) {
    try {
      const item = parseLiraProductPage_(productUrl);
      if (item && (!title || isRelevantTitleAuthorMatch_(item, title, author))) {
        results.push(item);
      }
    } catch (err) {
      if (trace) trace.productErrors.push({
        url: productUrl,
        error: err && err.message ? err.message : String(err)
      });
      // Egy hibás termékoldal ne állítsa le a többi találat feldolgozását.
    }
  });

  return dedupeItemsByIsbnOrSourceId_(results);
}

function searchLiraByIsbn_(isbn, trace) {
  const cleanIsbn = String(isbn || "").replace(/[^0-9Xx]/g, "");
  if (!cleanIsbn) return [];
  return searchLira_(cleanIsbn, "", "", trace);
}

function searchLiraByTitleAuthor_(title, author, trace) {
  const cleanTitle = String(title || "").trim();
  const cleanAuthor = normalizeAuthorForSearchOrder_(author);
  const query = buildLiraTitleSearchQuery_(cleanTitle);
  return searchLira_(query, cleanTitle, cleanAuthor, trace);
}
