/***********************************************************
 * LÍRA SEGÉDFÜGGVÉNYEK
 * Közvetlen Líra termékoldal feldolgozása.
 ***********************************************************/

function extractLiraOnlinePrice_(html) {
  const priceElement = extractFirstMatch_(
    html,
    /<div[^>]*class=["'][^"']*\bonline-price\b[^"']*["'][^>]*>([\s\S]*?)<\/div>/i
  );
  if (priceElement) {
    const elementMatch = String(priceElement).match(
      /([0-9]{1,3}(?:[ .][0-9]{3})+|[0-9]{3,})/
    );
    if (elementMatch) return elementMatch[1].replace(/[ .]/g, "");
  }

  const text = decodeHtmlEntities_(stripHtml_(html))
    .replace(/\u00a0/g, " ")
    .replace(/\s+/g, " ");
  const match = text.match(
    /online\s*ár\s*:[\s\S]{0,900}?([0-9]{1,3}(?:[ .][0-9]{3})+|[0-9]{3,})\s*Ft/i
  );
  return match ? match[1].replace(/[ .]/g, "") : "";
}

function parseLiraProductPage_(productUrl) {
  const productHtml = fetchPublisherHtml_(productUrl);
  const plainText = decodeHtmlEntities_(stripHtml_(productHtml))
    .replace(/\u00a0/g, " ")
    .replace(/\s+/g, " ");
  const title =
    extractFirstMatch_(productHtml, /<h1[^>]*>([\s\S]*?)<\/h1>/i) ||
    extractFirstMatch_(productHtml, /<meta property=["']og:title["'] content=["']([^"']+)/i);
  const author =
    extractFirstMatch_(productHtml, /<h2[^>]*>[\s\S]*?<a[^>]*>([\s\S]*?)<\/a>/i) ||
    extractFirstMatch_(productHtml, /Szerző:\s*[\s\S]{0,120}?<a[^>]*>([\s\S]*?)<\/a>/i);
  const publisher = extractFirstMatch_(productHtml, /Kiadó:\s*[\s\S]{0,120}?<a[^>]*>([\s\S]*?)<\/a>/i);
  const foundIsbn = (plainText.match(/ISBN:\s*([0-9Xx-]{10,20})/i) || [])[1] || "";
  const price = extractLiraOnlinePrice_(productHtml);
  const shipping = (plainText.match(/Szállítás:\s*([0-9]+\s*-\s*[0-9]+\s*munkanap)/i) || [])[1] || "";
  const imageUrl = extractFirstMatch_(productHtml, /<meta property=["']og:image["'] content=["']([^"']+)/i);

  if (!title) return null;
  return {
    source: "lira", sourceId: productUrl, title: title, authors: author,
    isbn: foundIsbn, publisher: publisher, coverUrl: imageUrl, language: "hu",
    price: price, availability: price ? "Rendelhető" : "Nincs adat",
    shipping: shipping, productUrl: productUrl
  };
}

/**
 * Líra keresési találati oldalból könyv-termékoldal URL-eket gyűjt.
 */
function extractLiraProductUrls_(html) {
  const matches = String(html || "").match(
    /(?:https:\/\/(?:www\.)?lira\.hu)?\/hu\/konyv\/[^"' <#?]+/ig
  ) || [];
  const seen = {};

  return matches.map(function(url) {
    const decodedUrl = decodeHtmlEntities_(url);
    return decodedUrl.indexOf("/hu/konyv/") === 0
      ? "https://www.lira.hu" + decodedUrl
      : decodedUrl;
  }).filter(function(url) {
    if (seen[url]) return false;
    seen[url] = true;
    return true;
  });
}

function buildLiraSearchUrl_(query) {
  return "https://www.lira.hu/hu/reszletes_kereso" +
    "?listorder=release_date&listdirection=desc" +
    "&listpagenumber=50&listcurrentpage=0&quick_search_txt=" +
    encodeURIComponent(String(query || "").trim());
}

function searchLira_(query, title, author) {
  if (!String(query || "").trim()) return [];

  const searchHtml = fetchPublisherHtml_(buildLiraSearchUrl_(query));
  const productUrls = extractLiraProductUrls_(searchHtml);
  const results = [];

  productUrls.slice(0, 8).forEach(function(productUrl) {
    try {
      const item = parseLiraProductPage_(productUrl);
      if (item && (!title || isRelevantTitleAuthorMatch_(item, title, author))) {
        results.push(item);
      }
    } catch (err) {
      // Egy hibás termékoldal ne állítsa le a többi találat feldolgozását.
    }
  });

  return dedupeItemsByIsbnOrSourceId_(results);
}

function searchLiraByIsbn_(isbn) {
  const cleanIsbn = String(isbn || "").replace(/[^0-9Xx]/g, "");
  if (!cleanIsbn) return [];
  return searchLira_(cleanIsbn, "", "");
}

function searchLiraByTitleAuthor_(title, author) {
  const cleanTitle = String(title || "").trim();
  const cleanAuthor = String(author || "").trim();
  const query = [cleanTitle, cleanAuthor].filter(function(value) { return value; }).join(" ");
  return searchLira_(query, cleanTitle, cleanAuthor);
}
